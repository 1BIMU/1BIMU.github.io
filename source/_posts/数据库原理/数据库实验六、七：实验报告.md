CREATE PROCEDURE "bupt2023210047"."register_user"(  
    IN p_username VARCHAR,  
    IN p_password VARCHAR,  
    IN p_email VARCHAR  
)  
AS  
DECLARE  
    v_count INTEGER;  
BEGIN  
    SELECT COUNT(*) INTO v_count FROM "用户基本信息表" WHERE username = p_username;  
    
    IF v_count > 0 THEN  
        RAISE NOTICE '用户名 % 已存在', p_username;  
    ELSE

    END IF;  
END

    v_count INTEGER;

    FROM "购物车基本信息表"  
    WHERE user_id = p_user_id AND product_id = p_product_id;

    PERFORM setval('"购物车基本信息表_cart_id_seq"', (SELECT COALESCE(MAX(cart_id), 0) FROM "购物车基本信息表"));

## 五、实验心得与总结  

#### 1. 实验收获与理论掌握  

通过本次实验，我深入理解了 GaussDB（兼容 PostgreSQL）中存储过程（Stored Procedure） 和触发器（Trigger）的核心概念与应用场景：  

- **业务逻辑下沉**：学会了如何将复杂的业务逻辑（如“批量修改分类价格”时的父子分类递归逻辑、“用户注册”时的存在性检查）封装在数据库层。这不仅减少了应用层的代码量，还减少了网络交互次数，提高了系统性能。  
    
- **数据完整性守护**：深刻体会到了触发器作为“最后一道防线”的作用。通过编写“价格保护”和“库存检查”触发器，我明白了如何在数据写入前（`BEFORE INSERT/UPDATE`）自动拦截非法数据，确保了数据的逻辑一致性。  
    
- **PL/pgSQL 语法**：熟悉了 GaussDB 的过程化 SQL 语法，包括变量声明（`DECLARE`）、逻辑判断（`IF...ELSE`）、异常抛出（`RAISE EXCEPTION`）以及游标和记录的操作。  
    

#### 2. 遇到的问题及解决方案  

在实验过程中，我遇到了一些具有挑战性的技术问题，但通过分析与调试均已解决，这些经历极大地锻炼了我的排错能力：  

- **DAS 环境下的 SQL 解析问题**：  
    
    - **问题描述**：在 DAS 的“SQL 查询”窗口直接执行包含分号 `;` 的存储过程代码时，报错 `subprogram body is not ended correctly`。原因是编辑器将存储过程内部的分号误判为语句结束符，导致代码被截断。  
        
    - **解决方案**：我学会了使用 DAS “库管理”界面中的**可视化“新建存储过程”功能**。该界面能正确处理代码块，避免了手动转义字符的繁琐，大大提高了开发效率。  
        
- **主键序列（Sequence）冲突问题**：  
    
    - **问题描述**：在测试触发器插入数据时，报错 `Duplicate key value violates unique constraint`。  
        
    - **原因分析**：这是因为手动插入数据或导入数据后，数据库的主键自增序列没有同步更新，仍然停留在旧值。  
        
    - **解决方案**：使用了 `SELECT setval(...)` 语句手动校准序列值，使其跳到当前最大 ID 之后，解决了冲突。  
        
- **外键约束（Foreign Key）理解**：  
    
    - **问题描述**：在使用不存在的 `category_id` 插入商品时，报错违反外键约束。  
        
    - **心得**：这让我直观地看到了关系型数据库的一致性保证机制。插入子表数据必须依赖于父表数据的存在，这是数据库设计的基石。  
        

#### 3. 实验建议与反思  

- **工具的选择**：华为云 DAS 虽然方便（免安装），但在处理复杂 SQL 块时对语法的解析比较严格。未来在开发复杂逻辑时，我会优先考虑使用图形化创建入口，或者在本地使用专门的客户端工具。  
    
- **防御性编程**：在编写存储过程时，应该像本次实验一样，多考虑边界情况（如库存不足、ID冲突），并给出友好的报错信息（`RAISE NOTICE`），这对于系统的健壮性至关重要。

import pyodbc  

# ================= 配置区域 =================  
DSN_NAME = "gaussdb"        # ODBC数据源名称  
DB_USER = "bupt2023210047"  # 数据库账号  
DB_PASSWORD = "bupt2023210047@" # 请在此处填入你的真实密码  

# ===========================================  

def get_connection():  
    """建立数据库连接"""  
    try:  
        conn_str = f"DSN={DSN_NAME};UID={DB_USER};PWD={DB_PASSWORD}"  
        conn = pyodbc.connect(conn_str)  
        return conn  
    except Exception as e:  
        print(f"[Error] Connection failed: {e}")  
        return None  

def print_table(cursor, title):  
    """打印表数据 (仅显示最后5条以供预览)"""  
    print(f"\n--- {title} ---")  
    sql = 'SELECT category_id, name FROM "商品分类基本信息表" ORDER BY category_id DESC LIMIT 5'  
    cursor.execute(sql)  
    
    rows = cursor.fetchall()  
    print(f"{'ID':<10} | {'Name'}")  
    print("-" * 30)  
    for row in rows:  
        print(f"{row.category_id:<10} | {row.name}")  
    print("-" * 30)  

def main():  
    print("[Info] Connecting to database...")  
    conn = get_connection()  
    if not conn:  
        return  
    
    cursor = conn.cursor()  

    try:  
        # 准备测试数据 (避免操作现有业务数据)  
        print("\n[Info] Preparing test data...")  
        cursor.execute('INSERT INTO "商品分类基本信息表" (name, parent_id) VALUES (?, ?)', ('Temp_Delete_Item', 0))  
        cursor.execute('INSERT INTO "商品分类基本信息表" (name, parent_id) VALUES (?, ?)', ('Temp_Update_Item', 0))  
        conn.commit()  
        
        # ================= 实验核心逻辑 =================  
        
        # 1. 打印所有记录  
        print_table(cursor, "1. Initial State")  

        # 2. 删除某一行  
        print("\n[Action] Deleting item 'Temp_Delete_Item'...")  
        cursor.execute('DELETE FROM "商品分类基本信息表" WHERE name = ?', ('Temp_Delete_Item',))  
        conn.commit()  
        print(f"Rows affected: {cursor.rowcount}")  

        # 3. 删除后打印  
        print_table(cursor, "2. State After Delete")  

        # 4. 修改数据  
        print("\n[Action] Updating 'Temp_Update_Item' to 'Updated_Item'...")  
        cursor.execute('UPDATE "商品分类基本信息表" SET name = ? WHERE name = ?', ('Updated_Item', 'Temp_Update_Item'))  
        conn.commit()  
        print(f"Rows affected: {cursor.rowcount}")  

        # 5. 修改后打印  
        print_table(cursor, "3. State After Update")  

        # 6. 插入新数据  
        print("\n[Action] Inserting new item 'New_Insert_Item'...")  
        cursor.execute('INSERT INTO "商品分类基本信息表" (name, parent_id) VALUES (?, ?)', ('New_Insert_Item', 0))  
        conn.commit()  

        # 7. 最终状态打印  
        print_table(cursor, "4. Final State")  

    except Exception as e:  
        print(f"\n[Error] Operation failed: {e}")  
        conn.rollback()  
    
    finally:  
        # 清理资源并断开连接 (对应 SQLFreeStmt, SQLDisconnect)  
        print("\n[Info] Cleaning up resources...")  
        cursor.close()  
        conn.close()  
        print("[Success] Experiment finished.")  

if __name__ == "__main__":  
    main()

## 五、实验心得  

#### 1. 遇到的问题和收获  

- **重要问题与处理：**  
    
    - **主键冲突报错（Unique Constraint Violation）：** 在运行 Python 脚本进行数据插入操作时，遇到了 `Duplicate key value violates unique constraint` 错误。这是因为数据库中的自增序列（Sequence）没有随着手动插入数据而更新，导致生成的 ID 与现有 ID 重复 。  
        
        - **处理方法：** 使用 SQL 语句 `SELECT setval(...)` 将对应表的主键序列重置为当前最大 ID 值，从而解决了冲突问题 。  
            
    - **ODBC 驱动版本匹配问题：** 在配置数据源时，需要注意操作系统与驱动版本的匹配。  
        - **处理方法：** 按照实验环境要求，在 64 位环境下正确安装并使用了 GaussDB Unicode (64位) 驱动，并确保 DSN 名称与代码中一致 。  
            
- **收获：**  
    
    - 掌握了使用 Python (pyodbc) 通过 ODBC 接口连接华为 GaussDB 的方法，理解了连接字符串（DSN, UID, PWD）的配置细节 。  
        
    - 通过编写增删改查（CRUD）代码，深入理解了应用程序如何通过 SQL 语句与数据库进行动态交互，实现了从“手动操作数据库”到“编程控制数据库”的转变 。  
        
    - 理解了数据库序列（Sequence）在主键自增中的关键作用，以及维护数据一致性的重要性。  
        
- **建议：**  
    
    - 建议在批量导入数据后，习惯性检查并校准序列值。  
        
    - 对于简单的数据库应用开发，推荐使用 Python，其代码量少且对 ODBC 接口的封装更加友好 。  
        

#### 2. 简述数据库驱动的概念  

数据库驱动（Database Driver）是一个软件组件或模块，它充当了应用程序与数据库管理系统（DBMS）之间的“翻译器”或中间件 。  

- **作用：** 它实现了特定的数据库连接标准（如 ODBC 或 JDBC），允许应用程序使用统一的 API（函数接口）来访问不同的数据库，而无需了解数据库底层的通信协议。  
    
- **示例：** 本实验中使用的 `GaussDB ODBC Driver` 就是驱动程序，它负责将 Python 发出的通用 ODBC 指令转换为 GaussDB 能够理解的指令 。  
    

#### 3. 简述 ODBC 开发应用的流程  

根据实验内容和 ODBC 标准，开发流程主要包含以下四个步骤 ：  

1. **初始化环境 (Initialization)：**  
    
    - 分配 ODBC 环境句柄（Environment Handle）和连接句柄（Connection Handle），为连接做准备。对应 API 如 `SQLAllocEnv` 和 `SQLAllocConnect` 。  
        
2. **建立连接 (Connection)：**  
    
    - 使用配置好的数据源名称（DSN）、用户名和密码，建立应用程序与数据库的物理连接。对应 API 如 `SQLConnect` 或 `SQLDriverConnect` 。  
        
3. **执行 SQL 操作 (Execution & Processing)：**  
    
    - 分配语句句柄（Statement Handle），发送 SQL 语句（增删改查）到数据库执行。对应 API 如 `SQLExecDirect` 。  
        
    - 如果是查询操作，还需要绑定结果列并提取数据（Fetching Data）。对应 API 如 `SQLFetch` 和 `SQLGetData` 。  
        
4. **断开连接与释放资源 (Termination)：**  
    
    - 操作完成后，释放语句句柄，断开数据库连接，并释放环境句柄，防止资源泄露。对应 API 如 `SQLFreeStmt`, `SQLDisconnect`, `SQLFreeEnv` 。
